#PCoA 
# -> Distanzmatrix (Aitchison) erstellen

def dist_matrix(filename="genus_example_merged_reads.csv"):

    import pandas as pd
    import numpy as np
    from skbio.stats.composition import clr
    from scipy.spatial.distance import pdist, squareform
    from skbio.stats.distance import DistanceMatrix
    
    # CSV laden
    df = pd.read_csv(filename)
    
    # Index auf Genus setzen
    df = df.set_index("name")
    
    # taxid entfernen
    df = df.drop(columns=["taxid"])
    
    # Transponieren: Samples als Zeilen
    df_t = df.T.fillna(0)
    
    # Zero replacement: kleinster nicht-null Wert / 2
    min_nonzero = df_t[df_t > 0].min().min()
    pseudo = min_nonzero / 2 if min_nonzero > 0 else 1e-6
    
    df_repl = df_t.replace(0, pseudo)
    
    # Closure (jede Zeile auf Summe 1)
    df_closed = df_repl.div(df_repl.sum(axis=1), axis=0)
    
    # CLR-Transformation
    df_clr = clr(df_closed)
    
    # Aitchison-Distanzmatrix (euclid im CLR-Raum)
    dist_array = pdist(df_clr, metric="euclidean")
    dist_matrix = DistanceMatrix(squareform(dist_array), ids=df_t.index)
    
    return dist_matrix

def PCoA(dist_matrix):
    from skbio.stats.ordination import pcoa
    import matplotlib.pyplot as plt
    
    # PCoA berechnen
    pcoa_res = pcoa(dist_matrix)
    
    # Koordinaten extrahieren
    coords = pcoa_res.samples
    
    # Plot
    plt.figure(figsize=(9,7))
    plt.scatter(coords["PC1"], coords["PC2"])
    
    for sample in coords.index:
        plt.text(coords.loc[sample, "PC1"],
                 coords.loc[sample, "PC2"],
                 sample,
                 fontsize=8)
    
    plt.xlabel(f"PC1 ({pcoa_res.proportion_explained[0]*100:.1f}%)")
    plt.ylabel(f"PC2 ({pcoa_res.proportion_explained[1]*100:.1f}%)")
    plt.title("PCoA")
    plt.show()

#PCoA 
# -> Distanzmatrix (Euclidian) für Wetter erstellen
# (eigene Funktion mit Euclidian, da Aitchison nicht mit negativen Werten kompativel ist, und der df nicht transponiert werden muss)

def dist_matrix_weather(filename="Copenhagen_weather.csv"):

    import pandas as pd
    import numpy as np
    from skbio.stats.composition import clr
    from scipy.spatial.distance import pdist, squareform
    from skbio.stats.distance import DistanceMatrix
    
    # CSV laden
    df = pd.read_csv(filename)
    
    # Index auf Genus setzen
    df = df.set_index("time")
    
    # fill Nans
    df_t = df.fillna(0)

    dist_array = pdist(df_t, metric="euclidean")
    dist_matrix = DistanceMatrix(squareform(dist_array), ids=df_t.index)

    return dist_matrix
